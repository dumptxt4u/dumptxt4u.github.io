	\section*{Short Answer Questions}
	
	\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=1.8em]
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[\frac{x-2}{x+2} + \frac{x+2}{x-2} = 4\]
			\item \[\frac{x-1}{x-2} + \frac{x-3}{x-4} = 3\frac{1}{3}\]
		\end{enumerate}
		
		\item 
		\[\frac{1}{x+1} - \frac{2}{x+2} = \frac{3}{x+3} - \frac{4}{x+4}\]
		
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[3^{x} + 3^{-x} = 2\]
			\item \[2^{x+1} + 4^{x} = 8\]
		\end{enumerate}
		
		\item 
		\[5^{x+1} + 5^{2-x} = 5^{3} + 1\]
		
		\item 
		\[x^{2/3} - x^{1/3} - 2 = 0\]
	\end{enumerate}

	\section*{Long Answer Questions}
	
	\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=1.8em, start=6]
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[(x^{2} + 3x + 2)^{2} - 8(x^{2} + 3x) - 4 = 0\]
			\item \[(x^{2} - 5x + 7)^{2} - (x-2)(x-3) = 1\]
		\end{enumerate}
		
		\item 
		\[x(x+2)(x+3)(x+5) = 72\]
		
		\item 
		\[x(x+2)(x^{2}-1) = -1\]
		
		\item 
		\[(x+1)(2x+3)(2x+5)(x+3) = 945\]
		
		\item 
		\[(x^{2} + x - 6)(x^{2} - 3x - 4) = 24\]
		
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[\sqrt{\frac{x}{1-x}} + \sqrt{\frac{1-x}{x}} = \frac{13}{6}\]
			\item \[\sqrt{\frac{4x-1}{4x+1}} - \sqrt{\frac{4x+1}{4x-1}} = \frac{8}{3},\quad x \neq \pm \frac{1}{4}\]
		\end{enumerate}
		
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[\sqrt{3x+4} = x\]
			\item \[\sqrt{3x+1} - \sqrt{x-1} = 2\]
		\end{enumerate}
		
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[\sqrt{x+1} + \sqrt{x-1} = 1\]
			\item \[\sqrt{x+3} + \sqrt{x-3} = 6\]
		\end{enumerate}
		
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[\sqrt{3x^{2} - 4x + 34} - \sqrt{3x^{2} - 4x - 11} = 5\]
			\item \[\sqrt{3x^{2} - 7x - 30} - \sqrt{2x^{2} - 7x - 5} = x - 5\]
		\end{enumerate}
		
		\item 
		\[12 + 9\sqrt{(x-1)(3x+2)} = 3x^{2} - x\]
		
		\item 
		\[\frac{\sqrt{x^{2} + 1} - \sqrt{x^{2} - 1}}{\sqrt{x^{2} + 1} + \sqrt{x^{2} - 1}} = \frac{1}{3}\]
		
		\item 
		\begin{enumerate}[label=(\roman*), leftmargin=2.2em]
			\item \[\left(x + \frac{1}{x}\right)^{2} - 2\left(x - \frac{1}{x} + 4\right) = 11\]
			\item \[12x^{4} - 56x^{3} + 89x^{2} - 56x + 12 = 0\]
		\end{enumerate}
		
		\item 
		\[(x^{2} + 2)^{2} + 8x^{2} = 6x(x^{2} + 2)\]
	\end{enumerate}
